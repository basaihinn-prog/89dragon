  🔒 What Happens Behind the Scenes

  When you activate a PIN code, the system:

  1. ✅ Validates user role - Only regular users can activate
  2. ✅ Checks if PIN system enabled - settings('payment_pin')
  3. ✅ Normalizes code - Converts to uppercase, trims whitespace
  4. ✅ Verifies PIN exists - In user's shop, status = active
  5. ✅ Checks cashier shift - Must be open
  6. ✅ Validates shop balance - Must have enough funds
  7. 💰 Starts transaction - Ensures all-or-nothing
  8. 💵 Deducts from shop - Shop balance decreases
  9. 💵 Adds to user - User balance increases
  10. 🎁 Calculates refund bonus - Based on rating/shop settings
  11. 📊 Updates shift stats - Money in, transfers count
  12. 📝 Logs transaction - Creates audit record
  13. 🗑 Deletes PIN - Security: prevents reuse
  14. 🔔 Fires event - Triggers MoneyIn listeners
  15. ✅ Commits transaction - Makes it permanent

  Swift/iOS

  func activatePinCode(token: String, pinCode: String, completion: @escaping (Result<[String: Any], Error>) 
  -> Void) {
      let url = URL(string: "https://bxbet.asia/api/mobile/voucher/activate")!
      var request = URLRequest(url: url)
      request.httpMethod = "POST"
      request.setValue("Bearer \(token)", forHTTPHeaderField: "Authorization")
      request.setValue("application/json", forHTTPHeaderField: "Content-Type")

      let body = ["code": pinCode]
      request.httpBody = try? JSONSerialization.data(withJSONObject: body)

      URLSession.shared.dataTask(with: request) { data, response, error in
          if let error = error {
              completion(.failure(error))
              return
          }

          guard let data = data else { return }

          if let json = try? JSONSerialization.jsonObject(with: data) as? [String: Any] {
              if json["success"] as? Bool == true {
                  completion(.success(json))
              } else {
                  let error = NSError(domain: "", code: 0, userInfo: [NSLocalizedDescriptionKey:
  json["error"] as? String ?? "Unknown error"])
                  completion(.failure(error))
              }
          }
      }.resume()
  }

  func activatePinCode(token: String, code: String) {
      let url = URL(string: "https://bxbet.asia/api/mobile/voucher/activate")!
      var request = URLRequest(url: url)
      request.httpMethod = "POST"
      request.setValue("Bearer \(token)", forHTTPHeaderField: "Authorization")
      request.setValue("application/json", forHTTPHeaderField: "Content-Type")
      request.httpBody = try? JSONEncoder().encode(["code": code])

      URLSession.shared.dataTask(with: request) { data, response, error in
          guard let data = data else { return }

          if let result = try? JSONDecoder().decode(PinResponse.self, from: data) {
              if result.success {
                  DispatchQueue.main.async {
                      // Update UI
                      self.showSuccess(amount: result.data.amountAdded,
                                     bonus: result.data.refundBonus,
                                     newBalance: result.data.newBalance)
                  }
              }
          }
      }.resume()
  }

  struct PinResponse: Codable {
      let success: Bool
      let message: String
      let data: PinData
  }

  struct PinData: Codable {
      let amountAdded: String
      let refundBonus: String
      let oldBalance: String
      let newBalance: String
      let currency: String
      let activatedAt: String

      enum CodingKeys: String, CodingKey {
          case amountAdded = "amount_added"
          case refundBonus = "refund_bonus"
          case oldBalance = "old_balance"
          case newBalance = "new_balance"
          case currency
          case activatedAt = "activated_at"
      }
  }

  Kotlin / Android

  suspend fun activatePinCode(token: String, pinCode: String): PinResponse {
      val client = OkHttpClient()
      val json = JSONObject().apply {
          put("code", pinCode)
      }

      val request = Request.Builder()
          .url("https://bxbet.asia/api/mobile/voucher/activate")
          .addHeader("Authorization", "Bearer $token")
          .addHeader("Content-Type", "application/json")
          .post(json.toString().toRequestBody("application/json".toMediaType()))
          .build()

      client.newCall(request).execute().use { response ->
          val responseBody = response.body?.string()
          val data = JSONObject(responseBody)

          if (data.getBoolean("success")) {
              val result = data.getJSONObject("data")
              // Update UI
              withContext(Dispatchers.Main) {
                  showSuccess(
                      amount = result.getString("amount_added"),
                      bonus = result.getString("refund_bonus"),
                      newBalance = result.getString("new_balance")
                  )
              }
              return@use data
          } else {
              throw Exception(data.getString("error"))
          }
      }
  }

  Common Errors

  | Error Message                                      | Meaning               | Solution                |
  |----------------------------------------------------|-----------------------|-------------------------|
  | Authentication required                            | No token provided     | Login first             |
  | Only regular users can activate vouchers           | Admin/cashier account | Use player account      |
  | PIN code system is not available                   | System disabled       | Contact admin to enable |
  | PIN code not found or does not belong to your shop | Invalid code          | Check code format       |
  | This PIN code has already been used                | Code used before      | Get new code            |
  | Cashier shift is not open                          | No active shift       | Contact support         |
  | Insufficient shop balance                          | Shop low on funds     | Contact support         |
  | Failed to activate PIN code                        | System error          | Check logs, retry       |
